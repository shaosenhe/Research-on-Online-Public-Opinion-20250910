# Modified tokenization function, returns word list and word frequency statistics for the document
def preprocess_text(text):
    text = replace_special_terms(text)
    words = jieba.cut(text)
    filtered_words = [word for word in words if word.strip() and word not in stopwords]
    word_counts = Counter(filtered_words)
    top_30_words = set([word for word, freq in word_counts.most_common(30)])
    final_words = [word for word in filtered_words if word in top_30_words]
    return final_words, word_counts

# Modified parallel tokenization function
def parallel_preprocess_texts(texts):
    with Pool(cpu_count()) as pool:
        results = pool.map(preprocess_text, texts)
    data_set = [result[0] for result in results]
    doc_word_counts = [result[1] for result in results]
    return data_set, doc_word_counts

# Modified filtering function, returns filtered data and word frequency statistics
def filter_low_frequency_words(data_set, original_doc_counts, min_docs=26):
    word_doc_count = defaultdict(int)
    for doc in data_set:
        unique_words = set(doc)
        for word in unique_words:
            word_doc_count[word] += 1

    filtered_data_set = []
    filtered_doc_counts = []
    for i, doc in enumerate(data_set):
        filtered_doc = [word for word in doc if word_doc_count[word] >= min_docs]
        filtered_data_set.append(filtered_doc)
        filtered_counts = {word: original_doc_counts[i][word] for word in filtered_doc if
                           word in original_doc_counts[i]}
        filtered_doc_counts.append(filtered_counts)

    return filtered_data_set, filtered_doc_counts

# Use multicore LDA model
def train_lda_model_multicore(num_topics, corpus, dictionary, num_words=30):
    print(f"\n开始训练主题数为 {num_topics} 的模型...")
    start_time = time.time()
    
    ldamodel = LdaMulticore(
        corpus,
        num_topics=num_topics,
        id2word=dictionary,
        passes=50,
        workers=cpu_count(),
        random_state=1,
        alpha=0.05,
        eta=0.005
    )
    
    elapsed_time = time.time() - start_time
    print(f"主题数 {num_topics} 的模型训练完成，耗时 {elapsed_time:.2f} 秒")
    print(f"主题数 {num_topics} 的主题 (显示前{num_words}个关键词):")
    for idx, topic in ldamodel.print_topics(num_topics=num_topics, num_words=num_words):
        print(f"主题 {idx}: {topic}")
    return ldamodel

# Calculate similarity between text and topics
def calculate_similarity(ldamodel, dictionary, texts, corpus, num_topics):
    topic_keywords = {}
    for i in range(num_topics):
        topic_words = [word for word, _ in ldamodel.show_topic(i, topn=50)]
        topic_keywords[i] = topic_words

    similarities = []
    for text in texts:
        # Convert text to bag-of-words model
        bow = dictionary.doc2bow(text)
        
        # Get text frequency vector
        text_freq = np.zeros(len(dictionary))
        for word_id, count in bow:
            text_freq[word_id] = count

        # Calculate similarity for each topic
        topic_similarities = []
        for i in range(num_topics):
            topic_weights = np.zeros(len(dictionary))
            for word in topic_keywords[i]:
                word_id = dictionary.token2id.get(word, -1)
                if word_id != -1:
                    topic_weights[word_id] = ldamodel.get_topics()[i][word_id]
            similarity = cosine_similarity([text_freq], [topic_weights])[0][0]
            topic_similarities.append(similarity)
        similarities.append(topic_similarities)

    return similarities, topic_keywords

# Calculate coherence
def coherence_score(num_topics, data_set, corpus, dictionary, num_words=30):
    ldamodel = train_lda_model_multicore(num_topics, corpus, dictionary, num_words)
    print(f"计算主题数 {num_topics} 的一致性分数...")
    start_time = time.time()
    ldacm = CoherenceModel(model=ldamodel, texts=data_set, dictionary=dictionary, coherence='c_v')
    coherence_score = ldacm.get_coherence()
    elapsed_time = time.time() - start_time
    print(f"主题数 {num_topics} 的 Coherence 分数: {coherence_score}，计算耗时 {elapsed_time:.2f} 秒\n")
    return coherence_score, ldamodel

# Calculate perplexity
def perplexity_score(ldamodel, corpus):
    print(f"计算困惑度...")
    start_time = time.time()
    perplex = ldamodel.log_perplexity(corpus)
    elapsed_time = time.time() - start_time
    print(f"困惑度: {perplex}，计算耗时 {elapsed_time:.2f} 秒\n")
    return perplex
